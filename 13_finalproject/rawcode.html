// colors

typedef struct {
  uint8_t r;
  uint8_t g;
  uint8_t b;
} RGBColor;

const int NUM_STOPS = 11;
const uint8_t stopPos[NUM_STOPS] = {0, 20, 55, 80, 110, 140, 165, 190, 215, 238, 255};
const RGBColor stopCol[NUM_STOPS] = {
  {0,   0,   0},
  {30,  0,   50},
  {60,  20,  160},
  {20,  60,  220},
  {0,   180, 200},
  {0,   200, 80},
  {180, 210, 0},
  {255, 180, 0},
  {255, 60,  40},
  {255, 120, 180},
  {255, 255, 255}
};

void mapToColor(uint8_t val, uint8_t *r, uint8_t *g, uint8_t *b) {
  for (int i = 0; i < NUM_STOPS - 1; i++) {
    if (val <= stopPos[i + 1]) {
      float t = (float)(val - stopPos[i]) / (float)(stopPos[i + 1] - stopPos[i]);
      *r = (uint8_t)(stopCol[i].r + t * ((float)stopCol[i+1].r - stopCol[i].r));
      *g = (uint8_t)(stopCol[i].g + t * ((float)stopCol[i+1].g - stopCol[i].g));
      *b = (uint8_t)(stopCol[i].b + t * ((float)stopCol[i+1].b - stopCol[i].b));
      return;
    }
  }
  *r = 255; *g = 255; *b = 255;
}

int xyToIndex(int x, int y) {
  if (x % 2 == 0) return x * ROWS + (ROWS - 1 - y);
  else             return x * ROWS + y;
}

//  camera s3 

bool initCamera() {
  camera_config_t config;
  config.ledc_channel = LEDC_CHANNEL_0;
  config.ledc_timer   = LEDC_TIMER_0;
  config.pin_d0       = Y2_GPIO;
  config.pin_d1       = Y3_GPIO;
  config.pin_d2       = Y4_GPIO;
  config.pin_d3       = Y5_GPIO;
  config.pin_d4       = Y6_GPIO;
  config.pin_d5       = Y7_GPIO;
  config.pin_d6       = Y8_GPIO;
  config.pin_d7       = Y9_GPIO;
  config.pin_xclk     = XCLK_GPIO;
  config.pin_pclk     = PCLK_GPIO;
  config.pin_vsync    = VSYNC_GPIO;
  config.pin_href     = HREF_GPIO;
  config.pin_sccb_sda = SIOD_GPIO;
  config.pin_sccb_scl = SIOC_GPIO;
  config.pin_pwdn     = PWDN_GPIO;
  config.pin_reset    = RESET_GPIO;
  config.xclk_freq_hz = 20000000;
  config.pixel_format = PIXFORMAT_GRAYSCALE;
  config.frame_size   = FRAMESIZE_QQVGA;
  config.fb_count     = 1;
  config.grab_mode    = CAMERA_GRAB_WHEN_EMPTY;
  config.fb_location  = CAMERA_FB_IN_PSRAM;

  esp_err_t err = esp_camera_init(&config);
  if (err != ESP_OK) {
    config.fb_location = CAMERA_FB_IN_DRAM;
    err = esp_camera_init(&config);
  }
  if (err != ESP_OK) return false;

  sensor_t *s = esp_camera_sensor_get();
  s->set_brightness(s, 0);
  s->set_contrast(s, 2);
  s->set_gainceiling(s, GAINCEILING_8X);
  if (mirrorHorizontal) s->set_hmirror(s, 1);
  return true;
}

// light display features

void frameToLEDs() {
  camera_fb_t *fb = esp_camera_fb_get();
  if (!fb) return;

  int blockW = fb->width  / COLS;
  int blockH = fb->height / ROWS;
  int total  = fb->width * fb->height;

  uint8_t frameMin = 255, frameMax = 0;
  for (int i = 0; i < total; i += 3) {
    if (fb->buf[i] < frameMin) frameMin = fb->buf[i];
    if (fb->buf[i] > frameMax) frameMax = fb->buf[i];
  }

  float range = (float)(frameMax - frameMin);
  if (range < 10.0) range = 10.0;

  for (int gy = 0; gy < ROWS; gy++) {
    for (int gx = 0; gx < COLS; gx++) {
      uint32_t sum = 0;
      int sx = gx * blockW;
      int sy = gy * blockH;
      for (int py = sy; py < sy + blockH; py++)
        for (int px = sx; px < sx + blockW; px++)
          sum += fb->buf[py * fb->width + px];

      uint8_t avg = sum / (blockW * blockH);

      float normalized = (float)(avg - frameMin) / range;
      if (normalized < 0.0) normalized = 0.0;
      if (normalized > 1.0) normalized = 1.0;

      float curved = pow(normalized, gammaVal);
      float target = curved * 255.0;
      if (invertImage) target = 255.0 - target;

      smoothGrid[gy][gx] = smoothGrid[gy][gx] * smoothFactor + target * (1.0 - smoothFactor);

      uint8_t val = (uint8_t)constrain((int)(smoothGrid[gy][gx] + 0.5), 0, 255);
      gridState[gy][gx] = val;

      uint8_t cr, cg, cb;
      mapToColor(val, &cr, &cg, &cb);
      strip.setPixelColor(xyToIndex(gx, gy), strip.Color(cr, cg, cb));
    }
  }

  strip.show();
  esp_camera_fb_return(fb);
}

//  HTTP 

esp_err_t page_handler(httpd_req_t *req) {
  httpd_resp_set_type(req, "text/html");
  return httpd_resp_send(req, HTML_PAGE, strlen(HTML_PAGE));
}

esp_err_t grid_handler(httpd_req_t *req) {
  char json[512];
  int pos = 0;
  pos += snprintf(json + pos, sizeof(json) - pos, "{\"f\":%d,\"g\":[", currentFPS);
  for (int y = 0; y < ROWS; y++)
    for (int x = 0; x < COLS; x++) {
      pos += snprintf(json + pos, sizeof(json) - pos, "%d", gridState[y][x]);
      if (y * COLS + x < NUM_LEDS - 1) json[pos++] = ',';
    }
  pos += snprintf(json + pos, sizeof(json) - pos, "]}");
  httpd_resp_set_type(req, "application/json");
  httpd_resp_set_hdr(req, "Access-Control-Allow-Origin", "*");
  return httpd_resp_send(req, json, pos);
}

esp_err_t settings_handler(httpd_req_t *req) {
  char buf[128];
  size_t buf_len = httpd_req_get_url_query_len(req) + 1;
  if (buf_len > 1 && buf_len <= sizeof(buf)) {
    httpd_req_get_url_query_str(req, buf, buf_len);
    char name[16], val[16];
    bool isToggle = (httpd_query_key_value(buf, "t", val, sizeof(val)) == ESP_OK);
    if (httpd_query_key_value(buf, "n", name, sizeof(name)) == ESP_OK) {
      if (isToggle) {
        if (strcmp(name, "invert") == 0) invertImage = !invertImage;
        if (strcmp(name, "mirror") == 0) {
          mirrorHorizontal = !mirrorHorizontal;
          sensor_t *s = esp_camera_sensor_get();
          s->set_hmirror(s, mirrorHorizontal ? 1 : 0);
        }
      }
      if (httpd_query_key_value(buf, "v", val, sizeof(val)) == ESP_OK) {
        int v = atoi(val);
        if (strcmp(name, "brightness") == 0) {
          ledBrightness = constrain(v, 5, 60);
          strip.setBrightness(ledBrightness);
        }
        if (strcmp(name, "smooth") == 0)
          smoothFactor = constrain(v, 0, 90) / 100.0;
        if (strcmp(name, "gamma") == 0)
          gammaVal = constrain(v, 10, 100) / 100.0;
      }
    }
  }
  char json[64];
  snprintf(json, sizeof(json), "{\"i\":%s,\"m\":%s}",
    invertImage ? "true" : "false",
    mirrorHorizontal ? "true" : "false");
  httpd_resp_set_type(req, "application/json");
  return httpd_resp_send(req, json, strlen(json));
}

void startWebServer() {
  httpd_config_t config = HTTPD_DEFAULT_CONFIG();
  config.server_port = 80;
  config.max_uri_handlers = 4;
  if (httpd_start(&web_httpd, &config) == ESP_OK) {
    httpd_uri_t u1 = {.uri="/",     .method=HTTP_GET, .handler=page_handler};
    httpd_uri_t u2 = {.uri="/grid", .method=HTTP_GET, .handler=grid_handler};
    httpd_uri_t u3 = {.uri="/set",  .method=HTTP_GET, .handler=settings_handler};
    httpd_register_uri_handler(web_httpd, &u1);
    httpd_register_uri_handler(web_httpd, &u2);
    httpd_register_uri_handler(web_httpd, &u3);
  }
  Serial.printf("Dashboard: http://%s\n", WiFi.localIP().toString().c_str());
}

void connectWiFi() {
  Serial.printf("Connecting to: %s", WIFI_SSID);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  WiFi.setSleep(false);
  int a = 0;
  while (WiFi.status() != WL_CONNECTED && a < 40) { delay(500); Serial.print("."); a++; }
  if (WiFi.status() == WL_CONNECTED)
    Serial.printf("\nConnected: http://%s\n", WiFi.localIP().toString().c_str());
  else
    Serial.println("\nWiFi failed. LEDs work offline.");
}

void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n=== LED Mirror 6x10 ===");

  for (int y = 0; y < ROWS; y++)
    for (int x = 0; x < COLS; x++)
      smoothGrid[y][x] = 0.0;

  strip.begin();
  strip.setBrightness(ledBrightness);
  strip.clear(); strip.show();

  for (int x = 0; x < COLS; x++) {
    for (int y = 0; y < ROWS; y++)
      strip.setPixelColor(xyToIndex(x, y), strip.Color(255, 255, 255));
    strip.show(); delay(80);
    for (int y = 0; y < ROWS; y++)
      strip.setPixelColor(xyToIndex(x, y), 0);
  }
  strip.clear(); strip.show();

  if (!initCamera()) {
    Serial.println("CAMERA FAILED");
    for (int i = 0; i < NUM_LEDS; i++)
      strip.setPixelColor(i, strip.Color(255, 0, 0));
    strip.show();
    while (true) delay(1000);
  }

  connectWiFi();
  if (WiFi.status() == WL_CONNECTED) startWebServer();
  Serial.println("Display online");
}

void loop() {
  unsigned long start = millis();
  frameToLEDs();
  unsigned long elapsed = millis() - start;
  currentFPS = elapsed > 0 ? 1000 / (int)elapsed : 0;

  static unsigned long lp = 0;
  if (millis() - lp > 3000) {
    Serial.printf("%lums (~%dfps) | http://%s\n",
      elapsed, currentFPS, WiFi.localIP().toString().c_str());
    lp = millis();
  }
}
